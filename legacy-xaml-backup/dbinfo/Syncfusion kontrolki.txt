        }

        private void LoadImionaFromDb()
        {
            var db = new AccessDbHelper();
            try
            {
                using (var conn = db.GetConnection())
                {
                    conn.Open();
                    var cmd = new OdbcCommand("SELECT S_imie FROM S_Imiona", conn);
                    using (var reader = cmd.ExecuteReader())
                    {
                        while (reader.Read())
                        {
                            if (reader["S_imie"] != DBNull.Value)
                                ImionaItems.Add(reader["S_imie"].ToString());
                        }
                    }
                }
            }
            catch { /* obsługa błędów wg potrzeb */ }
        }
